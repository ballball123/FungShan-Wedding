Google 試算表雙向保存與讀取設定教學

  1. 打開您的 Google 試算表 (Google Sheets)，建立一個空白試算表。
  2. 點選上方選單 「擴充功能」→「Apps Script」。
  3. 清空編輯器內的舊程式碼，貼上以下程式碼並按 Ctrl+S / Cmd+S 儲存：
Apps Script 程式碼（支援寫入及 JSONP 跨網域安全讀取）： 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var headers = payload.headers;
  var values = payload.values;
  var recordId = payload.recordId;
  var rawJson = payload.rawJson || '';

  if (sheet.getLastRow() === 0) {
    var fullHeaders = ['紀錄ID'].concat(headers).concat(['_原始資料JSON']);
    sheet.appendRow(fullHeaders);
    sheet.getRange(1, 1, 1, fullHeaders.length).setFontWeight('bold').setBackground('#18181b').setFontColor('#ffffff');
    sheet.setFrozenRows(1);
  }

  var rowData = [recordId].concat(values).concat([rawJson]);
  var dataRange = sheet.getDataRange().getValues();
  var existingRowIndex = -1;

  for (var i = 1; i < dataRange.length; i++) {
    if (String(dataRange[i][0]) === String(recordId)) {
      existingRowIndex = i + 1;
      break;
    }
  }

  if (existingRowIndex > 0) {
    sheet.getRange(existingRowIndex, 1, 1, rowData.length).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var callback = e && e.parameter && e.parameter.callback ? e.parameter.callback : '';
  var lastRow = sheet.getLastRow();
  var payload;

  if (lastRow < 2) {
    payload = JSON.stringify({ status: 'ok', records: [] });
  } else {
    var data = sheet.getDataRange().getDisplayValues();
    var headers = data[0];
    var records = [];

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (!row[0] && !row[2]) continue;
      var lastCell = String(row[row.length - 1] || '').trim();
      records.push({
        recordId: String(row[0] || ''),
        headers: headers,
        row: row,
        rawJson: lastCell.indexOf('{') === 0 ? lastCell : ''
      });
    }
    payload = JSON.stringify({ status: 'ok', records: records });
  }

  if (callback) {
    return ContentService.createTextOutput(callback + '(' + payload + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(payload)
    .setMimeType(ContentService.MimeType.JSON);
}
